// Covers 5 clients on one project: Super-Admin panel, Vendor panel, Styler/
// Consultant panel, the public website (Wedding-Ease-User-Interface), and the
// Viva chatbot (easebot: easebot-backend + Wedding-Ease-Viva-Chat).
//
// PROPOSED DRAFT — not deployed. Copy into firestore.rules yourself and
// deploy when ready; this repo's hooks intentionally block direct edits to
// firestore.rules during QA.
//
// Verified against actual app code (models/services), not guessed, for every
// field name below. Where a field name could not be verified with confidence
// it is called out in the accompanying summary, not silently assumed here.

service cloud.firestore {
  match /databases/{database}/documents {

    // ── Identity helpers ───────────────────────────────────────────────────
    function isAuthenticated() {
      return request.auth != null;
    }

    function isTeamMember() {
      return isAuthenticated() && exists(/databases/$(database)/documents/teamMember/$(request.auth.uid));
    }

    function getTeamMember() {
      return get(/databases/$(database)/documents/teamMember/$(request.auth.uid)).data;
    }

    function isAdmin() {
      return isTeamMember() && getTeamMember().userType == 'Admin';
    }

    function isStyler() {
      return isTeamMember() && getTeamMember().userType == 'Styler';
    }

    function isConsultants() {
      return isTeamMember() && getTeamMember().userType == 'Consultants';
    }

    // Any internal WeddingEase staff member (admin, styler, consultant).
    function isStaff() {
      return isAdmin() || isStyler() || isConsultants();
    }

    function isVendor() {
      return isAuthenticated() && exists(/databases/$(database)/documents/vendors/$(request.auth.uid));
    }

    function getProduct(productId) {
      return get(/databases/$(database)/documents/products/$(productId)).data;
    }

    // =========================================================================
    // SECTION A — shared identity documents
    // =========================================================================

    // 1. users — end-customer profile (main website + easebot share this doc).
    //    isVerified/isValidated are staff/server-controlled. subscription/
    //    tokenBalance/usage/plan are billing-tier state written server-side
    //    only by functions/services/subscriptionStateMachine.js and
    //    functions/theweddingbot/v1/account/getUsage.js — CONFIRMED to live
    //    as top-level fields on this same doc (not a subcollection), so they
    //    must be blocked here or a client can self-upgrade to any paid tier
    //    and grant themselves unlimited AI tokens with zero payment.
    match /users/{userId} {
      allow get: if isStaff() || (isAuthenticated() && request.auth.uid == userId);
      // list: UserService.getUserByEmail()/getUserByPhone() run
      // where('email'/'phone','==',x).limit(1) — used PRE-AUTH by signup
      // stale-account recovery, post-failed-login diagnostics, and the
      // forgot-password flow (AuthContext.tsx, SignInStep.tsx), so this
      // can't be restricted to isAuthenticated()/isStaff(). Bounded to
      // limit(1) so a single query can't dump the collection.
      // KNOWN TRADEOFF (flagged to owner, not fixed here — out of scope
      // for a rules-only change): this also lets anyone who already knows/
      // guesses a target email read that user's full profile, including
      // `forgotPasswordOtp` — a hash (not plaintext) of a 6-digit OTP, per
      // functions/mail.js `sendForgotPasswordOtp`, but a 6-digit numeric
      // space is trivially brute-forced offline once the hash is known.
      // Real fix (Cloud Functions change, not rules): move
      // forgotPasswordOtp/Expiry/Attempts off `users/{uid}` into their own
      // Admin-SDK-only collection (mirroring easebot's passwordResetOtps
      // pattern). Separately, functions/mail.js calls an undefined
      // `hashOtp`/`constantTimeEqualHex` — sendForgotPasswordOtp and
      // updateUserAccountPassword both throw at runtime today, so the
      // email-based reset flow is currently broken independent of rules.
      // list: also open to isStaff() (uncapped) — the styler panel's Users
      // tab (users.dart) list-queries `users` filtered by
      // `consultantId==<own uid>`, paginated via .limit(20)+cursor, which
      // request.query.limit<=1 was rejecting for every role including
      // staff.
      allow list: if isStaff() || request.query.limit <= 1;
      // get: `phone_*` branch — phone/WhatsApp-OTP signup (chai-sutra event
      // users) never establishes a Firebase Auth session (Twilio/WhatsApp
      // OTP, not Firebase Phone Auth — see AuthContext.verifyPhoneOTP), so
      // request.auth is null for the entire flow. ensureUserSession() reads
      // this doc back (getUser) to reuse an existing chat session; without
      // this it can't, and every login mints a new orphan session. Scoped to
      // the `phone_<ts>_<rand>` id namespace so Firebase-Auth-uid docs
      // (email/Google users) stay owner-only. NOTE (accepted tradeoff, see
      // owner): this lets an anonymous client read any phone user's profile
      // if they know/guess its doc id.
      allow get: if isStaff() || (isAuthenticated() && request.auth.uid == userId) ||
        userId.matches('phone_.*');
      // create blocklist NOTE: `isVerified`/`isValidated`/`verifiedAt` are
      // intentionally NOT blocked for the authenticated owner — the app sets
      // them client-side at the end of every signup verification (Google
      // new-user create AuthContext.tsx:613, email verify :1153, phone verify
      // :1186; verifyUserPhone/verifyUserEmail write {isVerified, isValidated,
      // verifiedAt} directly). Blocking them broke ALL signup methods with
      // "Missing or insufficient permissions", not just the chai-sutra phone
      // flow. Only the billing/token fields stay locked here — those are the
      // real self-upgrade-to-paid-tier / free-token risk. Verification is
      // already client-driven (OTP check runs client-side), so keeping these
      // out of the blocklist does not meaningfully weaken the trust model.
      // The anonymous `phone_*` branch stays STRICTER (still blocks the
      // verification flags too) since it has no auth at all.
      allow create: if isAdmin() || (isAuthenticated() && request.auth.uid == userId &&
        (!request.resource.data.keys().hasAny(['subscription', 'tokenBalance', 'usage', 'plan']))
      ) ||
        (userId.matches('phone_.*') &&
          !request.resource.data.keys().hasAny(['isVerified', 'isValidated', 'subscription', 'tokenBalance', 'usage', 'plan'])
        );
      // update: isStaff() branch added — booking_form.dart (styler panel)
      // writes `{'isfree_cons_used': true}` onto the CUSTOMER's users doc
      // inside a runTransaction() when a consultant books a free
      // consultation for them, so request.auth.uid != userId there. Same
      // sensitive-field blocklist as the owner branch — staff shouldn't be
      // able to grant themselves verification/subscription/token state by
      // editing a customer's doc through an unrelated booking flow.
      // update: `phone_*` branch is deliberately NARROWER than the owner
      // branch — an unauthenticated caller may only touch the
      // session/login bookkeeping keys ensureUserSession() and the phone
      // signin path actually write (createOrUpdateUser: {sessionId} /
      // {lastLoginAt} + the merge's lastUpdated/favourites). hasOnly (not the
      // blocklist) is critical: it blocks overwriting `phone`/`email`/`name`/
      // `password`, so an anonymous client that knows a phone doc id can't
      // repoint that account's phone number and hijack the OTP login.
      // owner-branch blocklist drops isVerified/isValidated (see create note
      // above): AuthContext's Google auto-verify (:660), email verify (:1153)
      // and phone verify (:1186) all updateDoc {isVerified, isValidated,
      // verifiedAt} on the owner's own doc. Billing/token fields stay locked.
      // The isStaff() branch KEEPS the full blocklist — a consultant editing a
      // customer's doc through the free-consultation booking flow must not be
      // able to flip that customer's verification state.
      allow update: if isAdmin() ||
        (isStaff() && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['isVerified', 'isValidated', 'subscription', 'tokenBalance', 'usage', 'plan'])) ||
        (isAuthenticated() && request.auth.uid == userId &&
          (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['subscription', 'tokenBalance', 'usage', 'plan']))
        ) ||
        (userId.matches('phone_.*') &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['sessionId', 'lastLoginAt', 'lastUpdated', 'favourites'])
        );
      allow delete: if isAdmin();

      // ── easebot (Viva) planning-tool data, private to the owning user ────
      // Intentionally NOT matched here: users/{uid}/subscription/{current}
      // (a second, possibly-legacy representation of the same billing state —
      // see summary) — server/Admin-SDK-managed only, stays unreachable from
      // any client.
      match /checklists/{checklistId} {
        allow read, write: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
      }
      match /savedItems/{itemId} {
        allow read, write: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
      }
      match /notifications/{notifId} {
        allow read, write: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
      }
      match /likedProducts/{prodId} {
        allow read, write: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
      }
      match /projects/{projectId} {
        allow read, write: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
      }
      match /reminders/{reminderId} {
        allow read, write: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
      }
      match /shoppingLists/{listId} {
        allow read, write: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
      }
      match /budget/{docId} {
        allow read, write: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
      }
      match /settings/{docId} {
        allow read, write: if isAdmin() || (isAuthenticated() && request.auth.uid == userId);
      }

      // ── easebot billing/metering state — Admin-SDK-only, no client access ──
      // usage/extrasTxns/subscription are written exclusively via the Admin
      // SDK (tokenMeter.ts, subscriptionStateMachine.ts) — verified end-to-end
      // 2026-07-10. No client ever reads/writes these; no rule is declared
      // here, so they fall through to the deny-all catch-all at the bottom of
      // this file.
    }

    // 2. teamMember — staff identity (Admin / Styler / Consultants / Vendor / Manager).
    //    Docs carry a `password` field (lib/models/teammodel.dart, surfaced in
    //    team_member.dart / team_member_details.dart) — read must NOT be a
    //    bare isAuthenticated(), since that lets any customer/vendor signup
    //    list() the entire staff roster and harvest every password. Storing
    //    plaintext-equivalent passwords in Firestore at all is a schema issue
    //    to raise separately; this at least stops the open list/read.
    match /teamMember/{memberId} {
      // read: opened to any authenticated user (owner decision — see
      // conversation). teamMember docs carry a real `password` field
      // (team_member.dart sets it on create/edit in the SU admin panel),
      // which rules can't hide field-by-field from an otherwise-allowed
      // read/list. Without this, package checkout hard-fails: PackageChecko
      // utPage.tsx calls TeamMemberService.getRandomConsultant()/
      // getRandomStyler() (which list-query teamMember by
      // userType+isActive+isAvailable) before writing the order, so a
      // permission-denied here aborts the whole purchase. Also needed for
      // getTeamMemberById()/getConsultantById() used to resolve an assigned
      // styler/consultant's name for booking/session display. Guest (pre-
      // login) free-consultation auto-assignment on the public home page
      // (Consultaiton.tsx) still can't read this and degrades gracefully to
      // an empty teamMemberId (wrapped in try/catch) — intentionally NOT
      // opened further to fully public given the password field.
      // REAL FIX (owner flagged, not applied here — Flutter/Cloud Functions
      // change, out of scope for a rules-only pass): stop storing password
      // on teamMember docs (the live Cloud Function path, TeamMember.js
      // createTeamMember, already does NOT store it — only the direct
      // client-side Flutter write in team_member.dart does).
      allow read: if isAuthenticated();
      allow write: if isAdmin();
    }

    // 3. vendors — vendor identity, doc ID == vendor's auth uid.
    //    Vendor can edit their own contact/profile fields, never the
    //    admin-controlled trust/verification/assignment fields.
    match /vendors/{vendorId} {
      allow read: if true; // public marketplace browsing
      allow create: if isAdmin();
      // update: `subcatids` moved out of the blocklist into its own
      // append-only check — vendor_form.dart's "Add Sub-Category" flow does
      // `vendors.doc(uid).update({'subcatids': arrayUnion([newId])})` right
      // after creating the subcategory doc itself (see subcategories.create
      // below), which the old blanket block on `subcatids` denied outright.
      // hasAll(old list) still blocks a vendor from REMOVING ids or
      // replacing the array wholesale — only appends are allowed.
      allow update: if isAdmin() || (isAuthenticated() && request.auth.uid == vendorId &&
        (!request.resource.data.diff(resource.data).affectedKeys().hasAny(
          ['isActive', 'level', 'aiStudio', 'aiDescription', 'password']
        )) &&
        (request.resource.data.subcatids == resource.data.get('subcatids', []) ||
          (request.resource.data.subcatids is list && request.resource.data.subcatids.hasAll(resource.data.get('subcatids', []))))
      );
      allow delete: if isAdmin();

      match /importLedger/{ledgerId} {
        allow read, write: if isAdmin();
      }
    }

    // =========================================================================
    // SECTION B — catalog
    // =========================================================================

    // 4. products
    match /products/{productId} {
      allow read: if true;
      allow create: if isAdmin() || (isVendor() && request.resource.data.vendorDocId == request.auth.uid);
      allow update: if isAdmin() || (isVendor() && resource.data.vendorDocId == request.auth.uid &&
        (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['vendorDocId']))
      );
      allow delete: if isAdmin() || (isVendor() && resource.data.vendorDocId == request.auth.uid);
    }

    // 5. variants — ownership follows the parent product's vendorDocId.
    match /variants/{variantId} {
      allow read: if true;
      // create: getAfter(), not get() — EMPIRICALLY VERIFIED via Firebase
      // Rules Unit Testing. product_form.dart's onSubmit() (regular product
      // form, AI Studio, and image-to-product flows all share this pattern)
      // creates the PRODUCT doc and its VARIANT doc(s) in the SAME
      // runTransaction()/batch. get()/getProduct() only ever sees the
      // pre-transaction snapshot, so for a brand-new product the product
      // doc doesn't exist yet from the rules engine's point of view —
      // .vendorDocId access fails and the whole atomic commit was denied
      // with a 403, even though the write was entirely legitimate.
      // getAfter() evaluates once all writes in the transaction/batch are
      // applied, so it correctly sees the sibling product write while still
      // rejecting a variant attached to a DIFFERENT vendor's product.
      allow create: if isAdmin() || (isVendor() && getAfter(/databases/$(database)/documents/products/$(request.resource.data.productId)).data.vendorDocId == request.auth.uid);
      allow update: if isAdmin() || (isVendor() &&
        getProduct(resource.data.productId).vendorDocId == request.auth.uid &&
        getProduct(request.resource.data.productId).vendorDocId == request.auth.uid
      );
      allow delete: if isAdmin() || (isVendor() && getProduct(resource.data.productId).vendorDocId == request.auth.uid);
    }

    // 6. subcategories — a vendor's product-save transaction rewrites the
    //    aggregate counters (totalProducts/minPrice/maxPrice/allData) on the
    //    subcategory as part of the SAME client transaction, so vendors need
    //    a narrow update allowance for just those fields; category metadata
    //    itself (name/image/etc.) stays admin-only.
    match /subcategories/{subcatId} {
      allow read: if true;
      // create: vendor_form.dart's "Add Sub-Category" dialog lets a vendor
      // introduce a brand-new subcategory (not just update an existing
      // one's counters, below) — it always sets isActive:false itself, so
      // this just enforces that a vendor can't create an already-live
      // subcategory that would immediately surface in the public catalog;
      // admin review/activation happens separately.
      allow create: if isAdmin() || (isVendor() && request.resource.data.isActive == false);
      allow delete: if isAdmin();
      // `detailTypes` added — ai_studio_product_page.dart/image_to_product_
      // page.dart conditionally write it onto the subcategory doc when an
      // AI-detected detail type is new to that subcategory.
      allow update: if isAdmin() || (isVendor() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['totalProducts', 'minPrice', 'maxPrice', 'allData', 'detailTypes']) &&
        request.resource.data.totalProducts is int && request.resource.data.totalProducts >= 0 &&
        request.resource.data.minPrice is number && request.resource.data.minPrice >= 0 &&
        request.resource.data.maxPrice is number && request.resource.data.maxPrice >= request.resource.data.minPrice
      );
    }

    // 7. offerings — global taxonomy, read-only to everyone else.
    match /offerings/{offeringId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // 8. tags — global lookup; vendors were already granted write historically.
    match /tags/{tagId} {
      allow read: if true;
      allow write: if isAdmin() || isVendor();
    }

    // =========================================================================
    // SECTION C — commerce: orders, vendor-side orders, deliveries, payments
    // =========================================================================

    // 9. orders — owner field verified as `uid` (lowercase) from OrderModel.
    //    Vendors legitimately read one order at a time (dispatch flow, via
    //    `.doc(vendorOrder.orderDocId).get()`) and need to touch
    //    orderProductData (in-transit quantities) but nothing else.
    //    `get` vs `list` split: a bare isVendor() (or isStyler()/isConsultants())
    //    OR-branch that doesn't reference resource.data lets Firestore serve
    //    an UNFILTERED list() of the entire collection — any one vendor
    //    account could otherwise dump every customer's order data platform-
    //    wide. Restricting `list` to staff+owner closes that; vendors still
    //    reach their own orders exactly the way the app does today, by known
    //    doc ID. (Residual: order IDs may be sequential per the settings/sets
    //    orderNo counter, so `get`-by-guessed-ID isn't fully closed without a
    //    schema change — e.g. a denormalized vendorIds array on the order —
    //    flagged in the summary, not something rules alone can fix.)
    //    CORRECTED after a real regression report: the earlier version of this
    //    rule blocked `isPaid`/`orderStatus` outright, assuming those were
    //    server/webhook-only (confirmed via functions/razorpay.js and
    //    functions/payu.js). That's true for the GATEWAY path, but it missed
    //    the app's DEFAULT path — src/services/orderService.ts's
    //    markOrderAsPaidAndProcessing(), called directly by CheckoutContainer
    //    whenever VITE_ACTIVATE_PAYMENT is not 'true' (currently the default):
    //    the CLIENT itself sets isPaid/paidAmount/paidOn/orderStatus/
    //    statusUpdatedAt/statusUpdatedBy on its own order, with no server
    //    verification at all. Same pattern in rejectOrderByUser/payNowOrder,
    //    which both write orderStatus with no isPaid involved. Blocking these
    //    broke checkout entirely ("Missing or insufficient permissions").
    //    orderStatus/paidAmount/paidOn/statusUpdatedAt/statusUpdatedBy are now
    //    unblocked for the owner. isPaid keeps a narrower guard: it may only
    //    move from not-true to true (never re-toggled or un-paid once set),
    //    so a customer can't repeatedly tamper with an already-paid order.
    //    This does NOT close the underlying trust gap — as designed today,
    //    the app lets a customer self-declare payment with zero server
    //    verification when the gateway is off. That's an app-architecture
    //    issue (needs a Cloud Function to verify real payment before trusting
    //    isPaid), not something a rules-only fix can solve without breaking
    //    the current checkout flow. totalAmount and the vendor/coupon/margin
    //    fields stay locked — no confirmed customer-side flow ever touches
    //    those after order creation.
    match /orders/{orderId} {
      allow get: if isStaff() || isVendor() || (isAuthenticated() && request.auth.uid == resource.data.uid);
      allow list: if isStaff() || (isAuthenticated() && request.auth.uid == resource.data.uid);
      allow create: if isAdmin() || isStaff() || (isAuthenticated() && request.auth.uid == request.resource.data.uid);
      allow update: if isAdmin() || isStyler() || isConsultants() ||
        (isAuthenticated() && request.auth.uid == resource.data.uid &&
          !request.resource.data.diff(resource.data).affectedKeys().hasAny([
            'totalAmount', 'totalVendorPrice', 'totalProfit', 'totalDiscount',
            'couponDiscount', 'couponCode', 'couponPackageOrderId', 'charges', 'uid'
          ]) &&
          (resource.data.isPaid == request.resource.data.isPaid ||
            (resource.data.isPaid != true && request.resource.data.isPaid == true))
        ) ||
        (isVendor() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['orderProductData']));
      allow delete: if isAdmin();
    }

    // 10. sessions — styling/consultation chat. Owner field verified `uId`.
    //     consultantId/stylerId are NOT staff-only — confirmed regression:
    //     PackageCheckoutPage.tsx:223 has the customer's own client write
    //     `{ consultantId, stylerId }` directly onto their session as part of
    //     normal package-purchase auto-assignment. Unblocked.
    match /sessions/{sessionId} {
      // `phone_*` branches — the phone/WhatsApp-OTP signup allocates a chat
      // session (ensureUserSession → ChatService.createSession) while
      // unauthenticated (no Firebase Auth session for phone users). Ownership
      // is keyed on the session's `uId` being in the `phone_` id namespace.
      // ACCEPTED TRADEOFF (owner): an anonymous client can read/write phone-
      // user sessions if it knows the (unguessable, addDoc-generated) session
      // id — the correct long-term fix is a custom-token sign-in so
      // request.auth carries the phone user's uid.
      // read is SPLIT into get/list on purpose: `get` (point read by known
      // session id) carries the phone_* branch, but `list` must NOT — a bare
      // resource.data.uId.matches('phone_.*') on `list` lets an anonymous
      // client range-query uId in ['phone_', 'phone_￿') and bulk-dump
      // every phone user's session (userName/budget/product interests). The
      // signup flow only ever does getSession() by id, never a list, so `list`
      // stays staff/owner-only.
      allow get: if isStaff() || (isAuthenticated() && resource.data.uId == request.auth.uid) ||
        (resource.data.uId is string && resource.data.uId.matches('phone_.*'));
      allow list: if isStaff() || (isAuthenticated() && resource.data.uId == request.auth.uid);
      allow create: if isStaff() || (isAuthenticated() && request.resource.data.uId == request.auth.uid) ||
        (request.resource.data.uId is string && request.resource.data.uId.matches('phone_.*'));
      allow update: if isStaff() ||
        (isAuthenticated() && resource.data.uId == request.auth.uid && request.resource.data.uId == request.auth.uid) ||
        (resource.data.uId is string && resource.data.uId.matches('phone_.*') &&
          request.resource.data.uId == resource.data.uId);
      allow delete: if isStaff() || (isAuthenticated() && resource.data.uId == request.auth.uid);

      match /messages/{messageId} {
        allow read, write: if isStaff() || (isAuthenticated() && get(/databases/$(database)/documents/sessions/$(sessionId)).data.uId == request.auth.uid);
      }
      match /internalChat/{chatId} {
        allow read, write: if isStaff(); // staff-only; no client UI ever reads this
      }
      match /deliveries/{deliveryId} {
        allow read, write: if isStaff() || (isAuthenticated() && get(/databases/$(database)/documents/sessions/$(sessionId)).data.uId == request.auth.uid);
      }
      match /checkOrders/{checkOrderId} {
        allow read, write: if isStaff() || (isAuthenticated() && get(/databases/$(database)/documents/sessions/$(sessionId)).data.uId == request.auth.uid);
      }
      match /stages/{stageId} {
        allow read, write: if isStaff() || (isAuthenticated() && get(/databases/$(database)/documents/sessions/$(sessionId)).data.uId == request.auth.uid);
      }
      match /timeline/{timelineId} {
        allow read, write: if isStaff() || (isAuthenticated() && get(/databases/$(database)/documents/sessions/$(sessionId)).data.uId == request.auth.uid);
      }
      match /cartHistory/{histId} {
        allow read, write: if isStaff() || (isAuthenticated() && get(/databases/$(database)/documents/sessions/$(sessionId)).data.uId == request.auth.uid);
      }
      match /stylerProducts/{prodId} {
        allow read, write: if isStaff() || (isAuthenticated() && get(/databases/$(database)/documents/sessions/$(sessionId)).data.uId == request.auth.uid);
      }
    }

    // 11. bookings — owner `uId`; assignment fields `stylerId`/`teamMemberId`.
    //     Create stays public (`true`) — free-consultation bookings can be
    //     started before/without a full account per current app behavior.
    //     The bare `resource.data.isFreeConsultation == true` branch has no
    //     auth dependency at all, so as a `list` rule it let ANY anonymous
    //     client dump every free-consultation booking platform-wide (name,
    //     email, phone, notes). Split to `get` (fetch-by-known-ID still works
    //     unauthenticated, matching whatever confirmation-page flow this
    //     supports) vs `list` (staff/owner only). Also locks staff-assignment
    //     and completion fields out of the customer's own update.
    match /bookings/{bookingId} {
      // get/list: checkFreeConsultationByEmail/getFreeConsultationByEmail
      // (bookingService.ts) query `bookings` filtered by
      // emailAtBooking+isFreeConsultation==true, called from BOTH
      // authenticated (AuthContext.tsx) and guest/unauthenticated flows
      // (Consultaiton.tsx, FreeConsultation.tsx) — the isFreeConsultation
      // exception now covers list too, not just get.
      allow get: if isStaff() || (isAuthenticated() && resource.data.uId == request.auth.uid) || resource.data.isFreeConsultation == true;

    //     design gap that rules alone cannot close (the secret and the
    //     access-control boundary are the same guessable value); `list` is
    //     still blocked so at least the collection can't be enumerated in
    //     bulk. STRONGLY recommend moving OTP verification server-side via a
    //     Cloud Function (mirroring passwordResetOtps) instead of hardening
    //     this further.
    match /phoneOtps/{phone} {
      allow get: if true;
      allow list: if isAdmin();
      // Verified exact payload in src/services/otpService.ts: `{ code, expiresAt }`,
      // nothing else — the code itself is stored in PLAINTEXT (not hashed, unlike
      // passwordResetOtps' sha256 approach), which rules cannot fix on the way in.
      // This shape check at least stops arbitrary extra fields being smuggled in.
      allow create, update: if request.resource.data.keys().hasOnly(['code', 'expiresAt']) &&
        request.resource.data.code is string &&
        request.resource.data.expiresAt is timestamp;
      allow delete: if true;
    }

    // =========================================================================
    // SECTION F — easebot / Viva chatbot (chats, notes, planning tools)
    // Shares `users/{uid}` with the rest of the project (see Section A) but
    // owns these collections outright. easebot-backend uses the Admin SDK for
    // passwordResetOtps and export/clear-history maintenance ops on `chats`,
    // so those two paths need no client rule beyond what's below.
    // =========================================================================

    // 37. chats(+messages) — Viva conversation threads. Owner field `userId`.
    match /chats/{threadId} {
      allow read, update, delete: if isAuthenticated() && resource.data.userId == request.auth.uid;
      allow create: if isAuthenticated() && request.resource.data.userId == request.auth.uid;

      match /messages/{messageId} {
        allow read, write: if isAuthenticated() && get(/databases/$(database)/documents/chats/$(threadId)).data.userId == request.auth.uid;
      }
    }

    // 38. sharedChats — intentionally public read-only snapshot (share-link
    //     feature), no owner field exists on the document. `list` is denied
    //     so the collection can't be enumerated; only a `get` by the known
    //     shareId (from the link) works. Treated as immutable once created.
    match /sharedChats/{shareId} {
      allow get: if true;
      allow list: if isAdmin();
      allow create: if isAuthenticated();
      allow update: if false;
      allow delete: if isAdmin();
    }

    // 39. notes(+comments) — owner field `ownerId`. Forward-compatible with
    //     the in-progress notes-sharing proposal: a doc may carry a
    //     `collaboratorEmails` array granting read/update to invited emails
    //     without requiring the field to exist on older documents.
    match /notes/{noteId} {
      allow read, update: if isAdmin() ||
        (isAuthenticated() && resource.data.ownerId == request.auth.uid) ||
        (isAuthenticated() && request.auth.token.email in resource.data.get('collaboratorEmails', []));
      allow create: if isAuthenticated() && request.resource.data.ownerId == request.auth.uid;
      allow delete: if isAdmin() || (isAuthenticated() && resource.data.ownerId == request.auth.uid);

      match /comments/{commentId} {
        allow read, write: if isAdmin() ||
          (isAuthenticated() && get(/databases/$(database)/documents/notes/$(noteId)).data.ownerId == request.auth.uid) ||
          (isAuthenticated() && request.auth.token.email in get(/databases/$(database)/documents/notes/$(noteId)).data.get('collaboratorEmails', []));
      }
    }

    // 40. noteFolders — owner field `ownerId`.
    match /noteFolders/{folderId} {
      allow read, write: if isAdmin() || (isAuthenticated() && resource.data.ownerId == request.auth.uid);
      allow create: if isAuthenticated() && request.resource.data.ownerId == request.auth.uid;
    }

    // 41. userImages — owner field `userId` (gallery uploads).
    match /userImages/{imageId} {
      allow read, write: if isAdmin() || (isAuthenticated() && resource.data.userId == request.auth.uid);
      allow create: if isAuthenticated() && request.resource.data.userId == request.auth.uid;
    }

    // 42. timelineEvents — owner field `ownerId`.
    match /timelineEvents/{eventId} {
      allow read, write: if isAdmin() || (isAuthenticated() && resource.data.ownerId == request.auth.uid);
      allow create: if isAuthenticated() && request.resource.data.ownerId == request.auth.uid;
    }

    // 43. feedback — owner field `userId`, nullable (guest feedback allowed).
    match /feedback/{feedbackId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    // 44. support_tickets (snake_case — easebot's own help-ticket system,
    //     separate from the main site's camelCase `supportTickets`).
    match /support_tickets/{ticketId} {
      allow create: if true; // guests allowed, userId is nullable
      allow read: if isAdmin() || (isAuthenticated() && resource.data.userId == request.auth.uid);
      allow update, delete: if isAdmin();
    }

    // =========================================================================
    // SECTION G — QA `*Test` collections (lib/shared/firebase.dart's `testMode`
    // switch, confirmed for admin/vendor/styler: every collection below gets a
    // `...Test`-suffixed twin when testMode is flipped on). CLAUDE.md
    // explicitly directs QA to "prefer *Test Firestore collections for any
    // write needed to exercise a flow" — with zero rules for these, every
    // *Test read/write would default-deny the moment testMode is used,
    // silently breaking the QA workflow this repo is under. Identity checks
    // (isAdmin/isStyler/isConsultants/isVendor) intentionally still resolve
    // against the REAL teamMember/vendors collections — testMode swaps out
    // business data, not who you're logged in as — so real staff/vendor/
    // customer accounts exercise these exactly like their prod counterparts.
    // Mirrors the same (post-fix) logic as each prod collection above.
    // =========================================================================

    match /usersTest/{userId} {
      // Mirrors the prod `users` owner blocklist: verification flags are
      // owner-settable (client-side signup verification), billing/token state
      // is not.
      allow read: if isStaff() || (isAuthenticated() && request.auth.uid == userId);
      allow create: if isAdmin() || (isAuthenticated() && request.auth.uid == userId &&
        (!request.resource.data.keys().hasAny(['subscription', 'tokenBalance', 'usage', 'plan'])));
      allow update: if isAdmin() || (isAuthenticated() && request.auth.uid == userId &&
        (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['subscription', 'tokenBalance', 'usage', 'plan'])));
      allow delete: if isAdmin();
    }
    match /teamMemberTest/{memberId} {
      allow read: if isAdmin() || (isAuthenticated() && request.auth.uid == memberId);
      allow write: if isAdmin();
    }
    match /vendorsTest/{vendorId} {
      allow read: if true;
      allow create: if isAdmin();
      allow update: if isAdmin() || (isAuthenticated() && request.auth.uid == vendorId &&
        (!request.resource.data.diff(resource.data).affectedKeys().hasAny(
          ['isActive', 'level', 'aiStudio', 'aiDescription', 'subcatids', 'password'])));
      allow delete: if isAdmin();
    }
    match /productsTest/{productId} {
      allow read: if true;
      allow create: if isAdmin() || (isVendor() && request.resource.data.vendorDocId == request.auth.uid);
      allow update: if isAdmin() || (isVendor() && resource.data.vendorDocId == request.auth.uid);
      allow delete: if isAdmin() || (isVendor() && resource.data.vendorDocId == request.auth.uid);
    }
    match /variantsTest/{variantId} {
      allow read, write: if isAdmin() || isVendor();
    }
    match /subcategoriesTest/{subcatId} {
      allow read: if true;
      allow write: if isAdmin() || isVendor();
    }
    match /tagsTest/{tagId} {
      allow read: if true;
      allow write: if isAdmin() || isVendor();
    }
    match /offeringsTest/{offeringId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /packagesTest/{packageId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /ordersTest/{orderId} {
      allow get: if isStaff() || isVendor() || (isAuthenticated() && request.auth.uid == resource.data.uid);
      allow list: if isStaff() || (isAuthenticated() && request.auth.uid == resource.data.uid);
      allow create: if isAdmin() || isStaff() || (isAuthenticated() && request.auth.uid == request.resource.data.uid);
      allow update: if isAdmin() || isStaff() || (isAuthenticated() && request.auth.uid == resource.data.uid) || isVendor();
      allow delete: if isAdmin();
    }
    match /packageOrdersTest/{orderId} {
      allow read, write: if isStaff() || (isAuthenticated() && resource.data.uid == request.auth.uid) || (isAuthenticated() && request.resource.data.uid == request.auth.uid);
    }
    match /transactionTest/{txId} {
      allow read, write: if isAdmin() || isVendor() || (isAuthenticated() && resource.data.uId == request.auth.uid) || (isAuthenticated() && request.resource.data.uId == request.auth.uid);
    }
    match /deliveryDetailsTest/{delId} {
      allow read, write: if isStaff() || isVendor() || (isAuthenticated() && resource.data.uId == request.auth.uid) || (isAuthenticated() && request.resource.data.vendorId == request.auth.uid);
    }
    match /venodrPurSaleTest/{saleId} {
      allow read: if true;
      allow write: if isAdmin() || isVendor();
    }
    match /vendorOrderTest/{orderId} {
      allow read, write: if isStaff() || (isAuthenticated() && resource.data.vendorId == request.auth.uid) || (isAuthenticated() && request.resource.data.vendorId == request.auth.uid);
    }
    match /sessionsTest/{sessionId} {
      allow read, write: if isStaff() || (isAuthenticated() && resource.data.uId == request.auth.uid) || (isAuthenticated() && request.resource.data.uId == request.auth.uid);

      match /{allSubcollections=**} {
        allow read, write: if isStaff() || isAuthenticated();
      }
    }
    match /bookingsTest/{bookingId} {
      allow create: if true;
      allow read, update, delete: if isStaff() || (isAuthenticated() && resource.data.uId == request.auth.uid);
    }
    match /wishlistTest/{wishlistId} {
      allow read, write: if isAdmin() || (isAuthenticated() && (resource == null || resource.data.uId == request.auth.uid) && (request.resource == null || request.resource.data.uId == request.auth.uid));
    }
    match /blogsTest/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /testimonialsTest/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /galleryTest/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /questionaireTest/{qId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // settings/sets update rule in Section D above now checks both 'sets'
    // and 'setsTest' doc IDs — no separate top-level match needed here.

    // 45. guests — easebot guest-quota tracking. Backend-only (tokenMeter.ts,
    //     Admin SDK); no client (frontend) ever reads/writes this directly.
    //     No rule declared — falls through to the deny-all catch-all below.

    // 46. payments — easebot payment/transaction records. Backend-only
    //     (subscriptionStateMachine.ts / paymentController.ts, Admin SDK);
    //     no client ever reads/writes this directly. No rule declared —
    //     falls through to the deny-all catch-all below.

    // 47. counters, invoices — easebot invoicing (invoiceService.ts, Admin
    //     SDK). Backend-only; confirmed zero frontend references anywhere in
    //     Wedding-Ease-Viva-Chat. No rule declared — falls through to the
    //     deny-all catch-all below.

    // 48. __health_check__ — synthetic liveness-probe collection
    //     (routes/health.ts GET /api/ready), never holds real data.
    match /__health_check__/{docId} {
      allow read: if true;
      allow write: if false;
    }

    // Explicit catch-all: anything not matched above is denied. Firestore
    // already default-denies unmatched paths, so this changes no behavior —
    // it documents that the omission of passwordResetOtps, users/{uid}/
    // subscription/**, and any other server-only path is intentional, not
    // an oversight for the next person editing this file.
    match /{document=**} {
      allow read, write: if false;
    }
  }
